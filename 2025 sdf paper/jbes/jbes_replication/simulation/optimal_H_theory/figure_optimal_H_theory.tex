% =============================================================================
% PANEL A: Objective level vs. score slope at theta_0 (from p1-score-logic.tex)
% =============================================================================
\begin{axis}[
  paperaxis,
  name=panelA,
  at={(0mm,0mm)},
  anchor=south west,
  width=44mm, height=40mm,
  xmin=-1.30, xmax=1.80, ymin=0, ymax=2.05,
  xtick={0, 0.62},
  xticklabels={$\theta_0$, $\theta_0^\dagger$},
  ytick=\empty, ymajorgrids=false,
  xlabel={Parameter $\theta$},
  ylabel={Population objective $\mathbb{E}[U_{v,H}(\theta)]$},
  ylabel style={yshift=0pt},
  xlabel style={yshift=1pt},
  title={Panel A: Level vs.\ slope at $\theta_0$},
  clip=false,
  legend style={at={(0.48,0.99)}, anchor=north}
]
  % Vertical reference at true theta_0
  \addplot[sharp plot, inkMuted, densely dotted, line width=0.6pt, forget plot]
    coordinates {(0,0) (0,1.45)};
  % H = 0 objective (tilted minimum at theta = 0.62)
  \addplot[inkMuted, line width=1.05pt, domain=-0.92:1.72, samples=70, smooth]
    {0.55*(x-0.62)^2 + 0.24};
  \addlegendentry{$H = 0$ (tilted)}
  % H = H_0* objective (higher level, centered minimum at theta_0 = 0)
  \addplot[accentRed, line width=1.15pt, domain=-1.18:1.18, samples=70, smooth]
    {0.55*x^2 + 0.78};
  \addlegendentry{$H = H_0^\ast$ (zero slope)}
  % Tangent slope at theta_0 for H = 0: y(0) = 0.55*0.62^2 + 0.24 = 0.4514; slope = -0.682
  \addplot[inkDark, line width=0.85pt, domain=-0.36:0.36]
    {0.4514 - 0.682*x};
  \addlegendentry{Slope $\nabla\mathbb{E}[U](\theta_0)$}

  % Key points
  \fill[inkDark] (axis cs:0, 0.4514) circle (1.3pt);
  \fill[accentRed] (axis cs:0, 0.78) circle (1.3pt);
  \fill[inkMuted] (axis cs:0.62, 0.24) circle (1.3pt);
  \addplot[sharp plot, inkFaint, densely dotted, line width=0.55pt, forget plot]
    coordinates {(0.62,0) (0.62,0.24)};

  % Bias arrow
  \draw[-{Stealth[length=1.4mm,width=1.1mm]}, inkDark, line width=0.6pt]
    (axis cs:0.03, 0.12) -- (axis cs:0.59, 0.12);

    at (axis cs:0.66, 0.12) {bias};
