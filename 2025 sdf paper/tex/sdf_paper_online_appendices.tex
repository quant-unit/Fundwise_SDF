\section{Proofs of Theorems T1--T8: Horizon Rescaling, Noise Tilt, and $U_{v,H}$}
\label{sec:appendix_theory}
% ==============================================================================

This appendix collects complete proofs of Theorems~T1--T8 stated in Section~\ref{sec:spatial_sdf_methodology} of the main paper. Throughout, $\theta$ lies in the positive-return admissible domain $\Theta$ where $G_u(\theta) = 1 + r_{f,u} + \alpha + \beta^\top F_u > 0$ on the relevant dates, so $\Psi_t(\theta) = \prod_{u=1}^t G_u(\theta)^{-1} > 0$ and $R^b_{a,b}(\theta) = \Psi_a(\theta)/\Psi_b(\theta) > 0$.

\subsection{Proof of Theorem T1 (Exact horizon rescaling identity)}
\label{sec:proof_T1}

For any unit (fund or constant-composition vintage portfolio) with common vintage origin $t_v^0$ and cash-flow stream $\{CF_{v,t}\}_{t=t_v^0}^{T_{\max}}$, the time-$\tau$ revalued pricing error at any date $\tau \in \mathcal{T}_{v,H}$ is
\[
\epsilon_{v,\tau}(\theta)
= \sum_{t=t_v^0}^{T_{\max}} \frac{\Psi_t(\theta)}{\Psi_\tau(\theta)} CF_{v,t}
= \frac{\Psi_{t_v^0}(\theta)}{\Psi_\tau(\theta)} \sum_{t=t_v^0}^{T_{\max}} \frac{\Psi_t(\theta)}{\Psi_{t_v^0}(\theta)} CF_{v,t}
= R^b_{t_v^0,\tau}(\theta)\, PV_v(\theta),
\]
where the factorization holds identically whether $\tau < t_v^0$, $\tau = t_v^0$, or $\tau > t_v^0$ because $\Psi_t(\theta)/\Psi_\tau(\theta) = (\Psi_{t_v^0}(\theta)/\Psi_\tau(\theta))\cdot(\Psi_t(\theta)/\Psi_{t_v^0}(\theta))$ pathwise on the positive-return domain. Averaging over $\tau \in \mathcal{T}_{v,H}$ yields
\[
\bar\epsilon_{v,H}(\theta)
= \frac{1}{|\mathcal{T}_{v,H}|} \sum_{\tau \in \mathcal{T}_{v,H}} \epsilon_{v,\tau}(\theta)
= \underbrace{\left(\frac{1}{|\mathcal{T}_{v,H}|} \sum_{\tau \in \mathcal{T}_{v,H}} R^b_{t_v^0,\tau}(\theta)\right)}_{A_{v,H}(\theta)\,>\,0} PV_v(\theta).
\]
In particular, $\bar\epsilon_{v,H}(\theta)$ is an exact path- and parameter-dependent scalar multiple of the inception NPV $PV_v(\theta)$. Because all dates $\tau \in \mathcal{T}_{v,H}$ share the common factor $PV_v(\theta)$, averaging across $\tau$ does \emph{not} average imperfectly correlated date-specific pricing errors; conditional on the factor path, all $\epsilon_{v,\tau}(\theta)$ are perfectly collinear with $PV_v(\theta)$. \qed

\subsection{Proof of Theorem T2 (Valid moment weights under (P) vs.\ (X))}
\label{sec:proof_T2}

Under inception pricing condition~(P), $\mathbb{E}[PV_v(\theta_0)\mid\mathcal{F}_{t_v^0}] = 0$. Therefore, for any integrable $\mathcal{F}_{t_v^0}$-measurable weight $z_v$ (including any pre-inception discounting factor $\Psi_{t_v^0}(\theta_0)/\Psi_\tau(\theta_0)$ with $\tau \le t_v^0$, or $z_v = 1$), the law of iterated expectations gives
\[
\mathbb{E}[z_v PV_v(\theta_0)] = \mathbb{E}\!\left[z_v \,\mathbb{E}[PV_v(\theta_0)\mid\mathcal{F}_{t_v^0}]\right] = 0.
\]
When $\tau > t_v^0$, the compounding weight $R^b_{t_v^0,\tau}(\theta_0) = \Psi_{t_v^0}(\theta_0)/\Psi_\tau(\theta_0)$ depends on public factor realizations after $t_v^0$ and is generally not $\mathcal{F}_{t_v^0}$-measurable. Taking expectations yields the Compounding-Horizon Wedge (repaired Lemma~1(3)):
\[
\mathbb{E}\!\left[\bar\epsilon_{v,H}(\theta_0)\,\middle|\,\mathcal{F}_{t_v^0}\right]
= \mathrm{Cov}\!\left(A_{v,H}(\theta_0),\, PV_v(\theta_0)\,\middle|\,\mathcal{F}_{t_v^0}\right)
=: b_{v,H},
\]
which is generically nonzero under~(P) alone whenever cash-flow pricing errors covary with realized benchmark returns over $\mathcal{T}_{v,H}$. Under the stronger path-exogeneity condition~(X), $\mathbb{E}[PV_v(\theta_0)\mid\mathcal{C}_v] = 0$, where $A_{v,H}(\theta_0)$ is $\mathcal{C}_v$-measurable; iterated expectations immediately yield $\mathbb{E}[A_{v,H}(\theta_0)PV_v(\theta_0)] = \mathbb{E}[A_{v,H}(\theta_0)\mathbb{E}[PV_v(\theta_0)\mid\mathcal{C}_v]] = 0$ at every horizon $H \ge 0$. \qed

\subsection{Proof of Theorem T3 (Signal-plus-noise decomposition of squared loss)}
\label{sec:proof_T3}

Let $\mathcal{G}_v$ contain $\mathcal{C}_v$, the vintage fund count $n_v$, and fund weights $\{w_{v,i}\}_{i=1}^{n_v}$ summing to one. Under constant-composition pooling, $PV_v(\theta) = \sum_{i=1}^{n_v} w_{v,i} PV_{v,i}(\theta)$. Suppose funds within vintage $v$ share conditional mean $\mu_v(\theta) = \mathbb{E}[PV_{v,i}(\theta)\mid\mathcal{G}_v]$, common conditional variance $\sigma_v^2(\theta) = \mathrm{Var}(PV_{v,i}(\theta)\mid\mathcal{G}_v)$, and common pairwise conditional covariance $c_v(\theta) = \mathrm{Cov}(PV_{v,i}(\theta), PV_{v,j}(\theta)\mid\mathcal{G}_v)$ for $i \neq j$. For equally weighted funds ($w_{v,i} = 1/n_v$),
\[
\mathbb{E}[PV_v(\theta)^2\mid\mathcal{G}_v]
= \mu_v(\theta)^2 + \mathrm{Var}(PV_v(\theta)\mid\mathcal{G}_v)
= \mu_v(\theta)^2 + c_v(\theta) + \frac{\sigma_v^2(\theta) - c_v(\theta)}{n_v}.
\]
(For general normalized fund-size weights $w_{v,i}$ with Herfindahl index $H_v = \sum_{i=1}^{n_v} w_{v,i}^2$, $1/n_v$ is replaced by $H_v$, giving $c_v(\theta) + H_v(\sigma_v^2(\theta) - c_v(\theta))$.) Multiplying by $A_{v,H}(\theta)^2$ (which is $\mathcal{G}_v$-measurable) and taking unconditional expectations yields
\[
Q_H^{\mathrm{L2}}(\theta)
= \mathbb{E}\!\left[A_{v,H}(\theta)^2 \mu_v(\theta)^2\right]
+ \mathbb{E}\!\left[A_{v,H}(\theta)^2 c_v(\theta)\right]
+ \mathbb{E}\!\left[A_{v,H}(\theta)^2 \frac{\sigma_v^2(\theta) - c_v(\theta)}{n_v}\right].
\]
At $\theta = \theta_0$ under~(X), $\mu_v(\theta_0) = 0$ almost surely, so both the signal term $\mathbb{E}[A_{v,H}(\theta_0)^2\mu_v(\theta_0)^2]$ and its gradient $2\mathbb{E}[A_{v,H}(\theta_0)^2\mu_v(\theta_0)\nabla_\theta\mu_v(\theta_0)]$ vanish. However, the variance terms depend on $\theta$ through both $A_{v,H}(\theta)^2$ and the discounted-payoff dispersion $\sigma_v^2(\theta), c_v(\theta)$, so $\nabla_\theta Q_H^{\mathrm{L2}}(\theta_0) \neq 0$ whenever $n_v < \infty$ or $c_v(\theta) > 0$. \qed

\subsection{Proof of Theorem T4 (Inception score tilt and Scale-Jacobian FOC decomposition)}
\label{sec:proof_T4}

Differentiating the sample L2 objective $Q_{V,H}^{\mathrm{L2}}(\theta) = V^{-1}\sum_{v=1}^V A_{v,H}(\theta)^2 PV_v(\theta)^2$ yields the exact two-term FOC decomposition:
\[
\nabla_\theta Q_{V,H}^{\mathrm{L2}}(\theta)
= \underbrace{\frac{2}{V}\sum_{v=1}^V A_{v,H}(\theta)^2 PV_v(\theta)\,\nabla_\theta PV_v(\theta)}_{\text{Term~(I): Inception Discounting}}
+ \underbrace{\frac{2}{V}\sum_{v=1}^V A_{v,H}(\theta)^2 PV_v(\theta)^2\,\nabla_\theta \log A_{v,H}(\theta)}_{\text{Term~(II): Scale-Jacobian Penalty}}.
\]
Now specialize to the single-exit deal model where fund $i$ in vintage $v$ makes deals $j$ with outlay $c_j > 0$ at $a_j \ge t_v^0$ and exit payoff $c_j R^b_{a_j,b_j}(\theta_0)(1+\varepsilon_{i,j})$ at $b_j > a_j$, with independent mean-zero shocks $\mathrm{Var}(\varepsilon_{i,j}\mid\mathcal{G}_v) = \nu_j > 0$ and $c_v \equiv 0$. At $\theta$, fund $i$'s pricing error is $PV_{v,i}(\theta) = \sum_j c_j \Psi_{t_v^0,a_j}(\theta)[\Psi_{a_j,b_j}(\theta)R^b_{a_j,b_j}(\theta_0)(1+\varepsilon_{i,j}) - 1]$, so its conditional variance at $\theta$ is
\[
\sigma_v^2(\theta) = \sum_j c_j^2 \nu_j \left(\frac{R^b_{a_j,b_j}(\theta_0)}{R^b_{t_v^0,b_j}(\theta)}\right)^2.
\]
Using $\nabla_\theta \log R^b_{t_v^0,b_j}(\theta) =: S_{t_v^0,b_j}(\theta)$ and $\nabla_\theta \log A_{v,H}(\theta) =: \bar S_{v,H}(\theta)$, differentiating $A_{v,H}(\theta)^2 \sigma_v^2(\theta) / n_v$ at $\theta = \theta_0$ (where $R^b_{t_v^0,b_j}(\theta_0) = R^b_{t_v^0,a_j}(\theta_0)R^b_{a_j,b_j}(\theta_0)$ cancels) gives
\[
\nabla_\theta Q_H^{\mathrm{L2}}(\theta_0)
= 2\,\mathbb{E}\!\left[n_v^{-2}\, A_{v,H}(\theta_0)^2 \sum_{i=1}^{n_v}\sum_j \frac{c_j^2 \nu_j}{R^b_{t_v^0,a_j}(\theta_0)^2}\left(\bar S_{v,H}(\theta_0) - S_{t_v^0,b_j}(\theta_0)\right)\right]
\]
(keeping $n_v^{-2}$ inside the expectation when fund counts $n_v$ vary across vintages). When deals start at $t_v^0$ ($R^b_{t_v^0,a_j}=1$), this simplifies to $2\,\mathbb{E}[n_v^{-1} A_{v,H}(\theta_0)^2 \sum_j c_j^2 \nu_j (\bar S_{v,H}(\theta_0) - S_{t_v^0,b_j}(\theta_0))]$.
At $H=0$, $A_{v,0}(\theta) \equiv 1$ so $\bar S_{v,0}(\theta_0) = 0$. Since $\partial_\alpha \log R^b_{t_v^0,b_j}(\theta_0) = \sum_{u=t_v^0+1}^{b_j} G_u(\theta_0)^{-1} > 0$ strictly, $\partial_\alpha Q_0^{\mathrm{L2}}(\theta_0) < 0$ strictly. For single-factor $\beta$, positive average excess market returns yield $S_{t_v^0,b_j}(\beta_0) > 0$ on average, so $\partial_\beta Q_0^{\mathrm{L2}}(\beta_0) < 0$, tilting $\beta_0^\dagger > \beta_0$ upward. As $H$ rises, $\bar S_{v,H}(\theta_0)$ increases from $0$ and eventually exceeds $S_{t_v^0,b_j}(\theta_0)$, reversing the sign of $\nabla_\theta Q_H^{\mathrm{L2}}(\theta_0)$ and pulling $\beta_H^\dagger$ downward. In the joint $(\alpha,\beta)$ model at $H=0$, first-order linearization around $\theta_0$ gives $\theta_0^\dagger - \theta_0 \approx -[\nabla^2 Q_0^{\mathrm{L2}}(\theta_0)]^{-1}\nabla Q_0^{\mathrm{L2}}(\theta_0)$; because $[\nabla^2 Q_0^{\mathrm{L2}}(\theta_0)]_{\alpha\beta} > 0$, the large negative $\alpha$-score induces an upward shift in $\alpha_0^\dagger$ that attenuates $\beta_0^\dagger$ downward, formalizing the qualitative observation in \citet[p.~522]{DLP12}. \qed

\subsection{Proof of Theorem T5 (Large-$\alpha$ attenuation at $H=0$ vs.\ coercivity at $H>0$)}
\label{sec:proof_T5}

Fix $\beta$ on the positive-return domain and let $\alpha \to +\infty$. Then $G_u(\alpha,\beta) \to +\infty$ for every month $u$, so $\Psi_{t_v^0,t}(\alpha,\beta) = \prod_{u=t_v^0+1}^t G_u(\alpha,\beta)^{-1} \to 0$ for every $t > t_v^0$, while $\Psi_{t_v^0,t_v^0}(\alpha,\beta) = 1$. Hence $PV_{v,i}(\alpha,\beta) \to CF_{v,i,t_v^0}$, giving
\[
\lim_{\alpha\to\infty} Q_0^{\mathrm{L2}}(\alpha,\beta) = \mathbb{E}[CF_{v,t_v^0}^2]
\]
(and under a predetermined origin where $CF_{v,t_v^0} = 0$, both $\lim_{\alpha\to\infty} Q_0^{\mathrm{L2}}(\alpha,\beta) = 0$ and $\lim_{\alpha\to\infty} U_{v,0}(\alpha,\beta) = 0$). Whenever lifetime residual variance $\mathbb{E}[PV_v(\theta_0)^2]$ exceeds $\mathbb{E}[CF_{v,t_v^0}^2]$, an unconstrained $H=0$ minimizer is pulled toward $\alpha \to +\infty$.

Conversely, suppose $H > 0$ so that $\mathcal{T}_{v,H}$ contains a date $\tau^\ast$ strictly after the first nonzero cash flow $t_v^{\mathrm{first}} \ge t_v^0$. Then for $t = t_v^{\mathrm{first}} < \tau^\ast$, the compounding factor $R^b_{t_v^{\mathrm{first}},\tau^\ast}(\alpha,\beta) = \prod_{u=t_v^{\mathrm{first}}+1}^{\tau^\ast} G_u(\alpha,\beta) \sim \alpha^{\tau^\ast - t_v^{\mathrm{first}}} \to +\infty$ dominates every later cash flow $t > t_v^{\mathrm{first}}$ (which grows at strictly lower degree in $\alpha$). Thus $|\bar\epsilon_{v,H}(\alpha,\beta)| \to +\infty$, implying $\lim_{\alpha\to\infty} Q_H^{\mathrm{L2}}(\alpha,\beta) = +\infty$. For $U_{v,H}$, since initial capital calls $CF_{v,i,t_v^{\mathrm{first}}} < 0$ have the same sign across funds active at the earliest date, their dominant pair products $CF_{v,i,t_v^{\mathrm{first}}}CF_{v,j,t_v^{\mathrm{first}}} > 0$ are strictly positive whenever at least two funds invest at $t_v^{\mathrm{first}}$, yielding $\lim_{\alpha\to\infty} U_{v,H}(\alpha,\beta) = +\infty$. \qed

\subsection{Proof of Theorem T6 (Infill consistency, $n_v \to \infty$)}
\label{sec:proof_T6}

(a) Fix $V$ and let $n_v \to \infty$ for each $v = 1,\dots,V$. Conditional on $\mathcal{G}_v$ and any common vintage shock, the constant-composition pool NPV converges in probability by a within-vintage law of large numbers to the vintage's realized conditional mean $\tilde\mu_v(\theta)$. Hence $V^{-1}\sum_{v=1}^V A_{v,H}(\theta)^2 PV_v(\theta)^2 \xrightarrow{p} V^{-1}\sum_{v=1}^V A_{v,H}(\theta)^2 \tilde\mu_v(\theta)^2$ uniformly on the compact domain $\Theta$. When common vintage shocks are absent ($\tilde\mu_v(\theta_0) = 0$ for all $v$) and $\theta_0$ uniquely sets the limit to zero on $\Theta$, $\hat\theta_H^{\mathrm{L2}} \xrightarrow{p} \theta_0$ at every fixed $H \ge 0$. If unspanned common vintage shocks $\tilde\mu_v(\theta_0) \neq 0$ are present, fixed-$V$ infill converges instead to the random minimizer of $V^{-1}\sum_{v=1}^V A_{v,H}(\theta)^2 \tilde\mu_v(\theta)^2$.
(b) Letting both $n_v \to \infty$ and $V \to \infty$ under the increasing-domain ULLN of Appendix~\ref{sec:proofs_asymptotics_H} eliminates the $O(1/n_v)$ term in Theorem~T3, yielding limit $\mathbb{E}[A_{v,H}(\theta)^2(\mu_v(\theta)^2 + c_v(\theta))]$. \qed

\subsection{Proof of Theorem T7 (Cross-fund pair-product $U_{v,H}$ consistency under bounded $n_v \ge 2$)}
\label{sec:proof_T7}

For any vintage $v$ with $n_v \ge 2$ funds and normalized fund weights $w_{v,i} > 0$ ($\sum_{i=1}^{n_v} w_{v,i} = 1$; $w_{v,i} = 1/n_v$ under EW, $w_{v,i} = s_{v,i}/\sum_k s_{v,k}$ under FW), define the cross-fund pair-product statistic
\[
U_{v,H}(\theta)
:= \frac{\sum_{i \neq j} w_{v,i} w_{v,j}\, \bar\epsilon_{v,i,H}(\theta)\, \bar\epsilon_{v,j,H}(\theta)}{\sum_{i \neq j} w_{v,i} w_{v,j}}
= A_{v,H}(\theta)^2 \frac{\left(\sum_{i=1}^{n_v} w_{v,i} PV_{v,i}(\theta)\right)^2 - \sum_{i=1}^{n_v} w_{v,i}^2 PV_{v,i}(\theta)^2}{1 - \sum_{i=1}^{n_v} w_{v,i}^2}.
\]
Let $\mathcal{G}_v$ include $\mathcal{C}_v$, the eligibility event $n_v \ge 2$, realized fund sizes $\{s_{v,i}\}$, and the realized vintage origin $t_v^0$. Suppose that conditional on $(\mathcal{G}_v, \text{common shock}_v)$, funds $i \neq j$ are independent with conditional mean $\mu_{v,i}(\theta) = \mathbb{E}[PV_{v,i}(\theta)\mid\mathcal{G}_v]$ and pairwise conditional covariance $c_v(\theta) = \mathrm{Cov}(PV_{v,i}(\theta), PV_{v,j}(\theta)\mid\mathcal{G}_v)$. Taking conditional expectations term by term for $i \neq j$:
\[
\mathbb{E}\!\left[PV_{v,i}(\theta)PV_{v,j}(\theta)\,\middle|\,\mathcal{G}_v\right]
= \mu_{v,i}(\theta)\mu_{v,j}(\theta) + c_v(\theta).
\]
Hence
\[
\mathbb{E}\!\left[U_{v,H}(\theta)\,\middle|\,\mathcal{G}_v\right]
= A_{v,H}(\theta)^2 \left(
  \frac{\sum_{i \neq j} w_{v,i} w_{v,j} \mu_{v,i}(\theta)\mu_{v,j}(\theta)}{\sum_{i \neq j} w_{v,i} w_{v,j}}
  + c_v(\theta)
\right),
\]
which reduces to $A_{v,H}(\theta)^2(\mu_v(\theta)^2 + c_v(\theta))$ whenever all funds in vintage $v$ share a common conditional mean $\mu_{v,i}(\theta) = \mu_v(\theta)$. Crucially, all own-fund diagonal variance terms $\sigma_v^2(\theta)$ cancel identically for every finite $n_v \ge 2$. Weighting vintages by $q_v > 0$ (kept inside the expectation when $q_v$ varies with $n_v$) and letting $V \to \infty$ under the uniform law of large numbers of Appendix~\ref{sec:proofs_asymptotics_H} gives
\[
Q_{V,H}^U(\theta) = \frac{1}{V}\sum_{v=1}^V q_v U_{v,H}(\theta)
\xrightarrow{p} \mathbb{E}\!\left[q_v A_{v,H}(\theta)^2 \bigl(\mu_v(\theta)^2 + c_v(\theta)\bigr)\right]
\]
uniformly on $\Theta$. Provided the limiting criterion has a unique minimizer $\theta_H^{\dagger,\infty} \in \Theta$, $\hat\theta_H^U \xrightarrow{p} \theta_H^{\dagger,\infty}$. In particular, when $c_v \equiv 0$, $\mu_v(\theta_0) = 0$ conditional on $\mathcal{G}_v$ (implied by (X) together with exogeneity of eligibility, fund sizes, and $t_v^0$), and $\theta_0$ uniquely minimizes $\mathbb{E}[q_v A_{v,H}(\theta)^2 \mu_v(\theta)^2]$ on $\Theta$, $\hat\theta_H^U \xrightarrow{p} \theta_0$ at every fixed horizon $H \ge 0$ with bounded $n_v \ge 2$. \qed

\subsection{Proof of Theorem T8 (Mean-moment multiple roots and persistent vintage shocks)}
\label{sec:proof_T8}

In a one-period model with risk-free gross return $R_f$, excess market return $F$, and PE payoff $(R_f + \beta_{\mathrm{PE}} F)N$ with $\mathbb{E}[N\mid F] = 1$, the single-factor mean pricing moment with $\alpha = 0$ is
\[
m(\beta) = \mathbb{E}\!\left[\frac{R_f + \beta_{\mathrm{PE}} F}{R_f + \beta F} - 1\right]
= (\beta_{\mathrm{PE}} - \beta)\,\underbrace{\mathbb{E}\!\left[\frac{F}{R_f + \beta F}\right]}_{g(\beta)}.
\]
Whenever $\mathbb{E}[F] > 0$ and $\mathbb{P}(F < 0) > 0$, $g(0) = \mathbb{E}[F]/R_f > 0$ while $g(\beta) \to -\infty$ as $\beta$ approaches the upper positivity boundary, so there exists a unique growth-optimal root $\beta^\ast > 0$ solving $g(\beta^\ast) = 0$ in addition to the structural root $\beta_{\mathrm{PE}}$. For example, with $R_f = 1$ and $F \in \{-0.4, +0.6\}$ equally likely, $\beta_{\mathrm{PE}} = 1$ and $\beta^\ast = 5/12$ both satisfy $m(\beta)=0$ on the positive-return domain, so a ``smallest-root'' rule selects $5/12 \neq 1$. Under persistent vintage shocks ($c_v > 0$, $\rho_W > 0$), any instrument $z_v$ constructed from neighboring vintages' cash flows correlates with $PV_v(\theta_0)$; valid IV moments $\mathbb{E}[z_v PV_v(\theta_0)] = 0$ therefore require strictly exogenous path instruments $z_v = h(\mathcal{C}_v)$ (valid under (X), or $z_v = 1$ under (P) alone) or $h$-block neighbor-excluding instruments. \qed

\section{Increasing-Domain Asymptotics, Fixed-$b$ SHAC Inference, and Full Step 4 Monte Carlo Exhibits}
\label{sec:proofs_asymptotics_H}
% ==============================================================================

\subsection{Regularity conditions, consistency, and asymptotic normality}

Fix a horizon $H \ge 0$, and let $Q_{V,H}(\theta) = V^{-1}\sum_{v=1}^V \ell_{v,H}(\theta)$ denote the sample criterion over $V$ vintages, where $\ell_{v,H}(\theta) = \bar\epsilon_{v,H}(\theta)^2$ for L2 on vintage portfolios and $\ell_{v,H}(\theta) = q_v U_{v,H}(\theta)$ for the pair-product estimator (with $v$ ranging over eligible $n_v \ge 2$ vintages). Define the population objective $Q_H(\theta) = \mathbb{E}[\ell_{v,H}(\theta)]$ and target $\theta_H^\dagger = \arg\min_{\theta\in\Theta} Q_H(\theta)$.

\begin{assume}[Increasing-domain regularity conditions]
\label{ass:asymptotics_H}
\begin{enumerate}
  \item \textbf{Compact admissible domain:} $\Theta \subset \mathbb{R}^p$ is compact with $\min_{t} G_t(\theta) \ge \underline{g} > 0$ for all $\theta \in \Theta$ (in implementation, $\underline{g}=0.08$, $\beta_{\mathrm{MKT}}\in[-1,4]$, $\alpha\in[-2\%,2\%]$/month), and $\theta_H^\dagger$ is the unique global minimizer of $Q_H(\theta)$ on $\Theta$.
  \item \textbf{Smoothness and envelope bounds:} $\ell_{v,H}(\theta)$ is twice continuously differentiable on $\Theta$ with $\sup_{\theta\in\Theta}|\ell_{v,H}(\theta)|$, $\sup_{\theta\in\Theta}\|\nabla_\theta\ell_{v,H}(\theta)\|$, and $\sup_{\theta\in\Theta}\|\nabla_{\theta\theta}^2\ell_{v,H}(\theta)\|$ possessing uniformly bounded $(2+\delta)$ moments ($\delta > 0$).
  \item \textbf{Increasing-domain vintage mixing:} Vintages lie on an expanding calendar lattice ($V \to \infty$) with bounded per-vintage fund counts and bounded effective lifetimes, such that $\{\ell_{v,H}(\theta), \nabla_\theta\ell_{v,H}(\theta), \nabla_{\theta\theta}^2\ell_{v,H}(\theta)\}$ satisfy the near-epoch dependence / mixing conditions of \citet{JP09,JP12}.
  \item \textbf{Interior optimum and nonsingular curvature (for CLT):} $\theta_H^\dagger \in \mathrm{int}(\Theta)$, $\mathcal{H}_H := \mathbb{E}[\nabla_{\theta\theta}^2 \ell_{v,H}(\theta_H^\dagger)]$ is positive definite, and $\Lambda_H := \lim_{V\to\infty}\mathrm{Var}(V^{-1/2}\sum_{v=1}^V s_{v,H}(\theta_H^\dagger))$ is positive definite, where $s_{v,H}(\theta) := \nabla_\theta\ell_{v,H}(\theta)$.
\end{enumerate}
\end{assume}

\begin{proposition}[Consistency and asymptotic normality around $\theta_H^\dagger$]
\label{prop:proof_cons_clt}
Under Assumption~\ref{ass:asymptotics_H}(1)--(3), $\hat\theta_H \xrightarrow{p} \theta_H^\dagger$ as $V \to \infty$. Under Assumption~\ref{ass:asymptotics_H}(1)--(4),
\[
\sqrt{V}(\hat\theta_H - \theta_H^\dagger) \xrightarrow{d} \mathcal{N}(0,\Sigma_H),
\qquad
\Sigma_H = \mathcal{H}_H^{-1}\Lambda_H \mathcal{H}_H^{-1}.
\]
\end{proposition}

\begin{proof}
\emph{Consistency.} By Assumption~\ref{ass:asymptotics_H}(1)--(3) and the random-field uniform law of large numbers \citep{JP09,PP97}, $\sup_{\theta\in\Theta}|Q_{V,H}(\theta) - Q_H(\theta)| \xrightarrow{p} 0$. Compactness of $\Theta$, continuity of $Q_H$, and unique identification of $\theta_H^\dagger$ yield $\hat\theta_H \xrightarrow{p} \theta_H^\dagger$ by \citet[Theorem~2.1]{NM94}.

\emph{Asymptotic normality (score centering and integrated Hessian).} Since $\theta_H^\dagger \in \mathrm{int}(\Theta)$ and $\hat\theta_H \xrightarrow{p} \theta_H^\dagger$, the sample first-order condition $\nabla_\theta Q_{V,H}(\hat\theta_H) = 0$ holds with probability approaching one. Crucially, the population score is centered at the criterion target $\theta_H^\dagger$---i.e., $\mathbb{E}[s_{v,H}(\theta_H^\dagger)] = \nabla_\theta Q_H(\theta_H^\dagger) = 0$---rather than at $\theta_0$ (for L2 with finite $n_v$, $\mathbb{E}[s_{v,H}(\theta_0)] \neq 0$ by Theorem~T4). Applying the fundamental theorem of calculus to each component of $\nabla_\theta Q_{V,H}$ along the line segment $\theta(u) = \theta_H^\dagger + u(\hat\theta_H - \theta_H^\dagger)$, $u\in[0,1]$, yields the exact integrated-Hessian expansion
\[
0 = \nabla_\theta Q_{V,H}(\hat\theta_H)
  = \nabla_\theta Q_{V,H}(\theta_H^\dagger)
  + \left(\int_0^1 \nabla_{\theta\theta}^2 Q_{V,H}(\theta(u))\,du\right)(\hat\theta_H - \theta_H^\dagger).
\]
By Assumption~\ref{ass:asymptotics_H}(2)--(3), $\sup_{\theta\in\Theta}\|\nabla_{\theta\theta}^2 Q_{V,H}(\theta) - \nabla_{\theta\theta}^2 Q_H(\theta)\| \xrightarrow{p} 0$, and continuity of $\nabla_{\theta\theta}^2 Q_H$ at $\theta_H^\dagger$ implies $\bar{\mathcal{H}}_{V,H} := \int_0^1 \nabla_{\theta\theta}^2 Q_{V,H}(\theta(u))\,du \xrightarrow{p} \mathcal{H}_H$. By Assumption~\ref{ass:asymptotics_H}(3)--(4) and the increasing-domain random-field CLT of \citet{JP12}, $\sqrt{V}\nabla_\theta Q_{V,H}(\theta_H^\dagger) = V^{-1/2}\sum_{v=1}^V s_{v,H}(\theta_H^\dagger) \xrightarrow{d} \mathcal{N}(0,\Lambda_H)$. Slutsky's theorem yields $\sqrt{V}(\hat\theta_H - \theta_H^\dagger) = -\bar{\mathcal{H}}_{V,H}^{-1}\sqrt{V}\nabla_\theta Q_{V,H}(\theta_H^\dagger) \xrightarrow{d} \mathcal{N}(0,\mathcal{H}_H^{-1}\Lambda_H\mathcal{H}_H^{-1})$.
\end{proof}

\subsection{LLSW bandwidth, fixed-$b$ critical values, and boundary conventions}

In finite vintage panels ($V = 23$--$27$), classical HAC inference with normal critical values severely over-rejects because the Bartlett estimator $\hat\Lambda_H = V^{-1}\sum_{v,w} K(|v-w|/S)\,\hat s_{v,H}\hat s_{w,H}^\top$ is pulled toward zero by the sample FOC $\sum_v \hat s_{v,H} = 0$. Following \citet{LLSW18} and \citet{KV05}, we set the Bartlett truncation lag to $S = \lceil 1.3\sqrt{V}\rceil$ ($S=7$ for $V\in\{23,25,27\}$) and evaluate $t$- and Wald statistics against the fixed-$b$ limit distribution at $b = S/V$ (two-sided 5\% critical values $2.790$ for $V=25$, $2.867$ for $V=23$, and $2.728$ for $V=27$, computed via the conditional method with $10^6$ draws). Whenever an estimate lies on an active boundary constraint of $\Theta$, interior asymptotic normality does not apply; such cells are marked with $\dagger$ and their standard errors are suppressed. Because L1-loss estimates sit at sample kinks where numerical Hessians shrink with step size and the delete-one jackknife is inconsistent for median-type functionals \citep{Efron79}, L1 estimates are treated as descriptive only with no SEs or tests.
